{%- comment -%} Maps an app's category to one of the five stamp series. Usage: {% include series.html app=app %} → series_key, series_label (label in the page's language) {%- endcomment -%} {%- assign _c = include.app.category -%} {%- case _c -%} {%- when "health" or "fitness" -%}{%- assign series_key = "health" -%} {%- when "finance" -%}{%- assign series_key = "money" -%} {%- when "games" -%}{%- assign series_key = "games" -%} {%- when "developer-tools" or "business" or "graphics" or "utilities" -%}{%- assign series_key = "tools" -%} {%- else -%}{%- assign series_key = "plan" -%} {%- endcase -%} {%- assign _st = site.data.i18n[page.lang] | default: site.data.i18n.en -%} {%- assign series_label = _st.series[series_key] -%}
